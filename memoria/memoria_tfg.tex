\chapter{Introducción}\label{cap:introduccion}
El electromagnetismo computacional y el análisis mediante elementos finitos (FEM por sus siglas en inglés) son herramientas fundamentales para obtener soluciones numéricas a problemas físicos complejos o sin soluciones analíticas.Uno de los pilares de FEM es la selección de funciones de base adecuadas que eviten la aparición de soluciones espurias, aseguran la continuidad tangencial y permiten la discontinuidad en la componente normal. Este trabajo se centra en la implementación en Julia de un conjunto concreto de esas funciones de base, el elemento hexaédrico curl-conforme de Nédélec de segundo orden
propuesto en \cite{amor2023hexahedral}, y verifica la implementación resolviendo el mismo problema de referencia (una cavidad semillena) frente a la implementación en MATLAB proporcionada en \cite{amor2022zenodo}
%--------------------------------------------------------------------------

\section{Motivación}\label{sec:motivacion}
El Grupo de Radiofrecuencia, Electromagnetismo, Microondas y Antenas
(GREMA) de la Universidad Carlos III de Madrid ha contribuido al electromagnetismo computacional con especial énfasis en FEM desarrollando herramientos como HOFEM (higher Order
Finite Element Method) \cite{garciadonoro2015hofem}. HOFEM discretiza
la ecuación de onda vectorial de doble rotacional mediante tetrahedros y prismas
curl-conformes isoparamétricos de la primera familia de Nedelec y se puede utilizar mediante un simulador gráfico tanto en Linux como Windows.

El simulador está escrito en Fortran y utilizando el paradigma de la programación orientada a objetos con el objetivo de hacerlo mantenible, sin embargo , en muchas ocasiones en la programación científica se utiliza un lenguaje de alto nivel(MATLAB o python) que permita realizar un prototipo o prueba de concepto y luego utilizar las funcionalidades de Fortran como la alta eficiencia, portabilidad o capacidad de programación en paralelo para crear el producto definitivo.

Con el objetivo de acercar ambas aproximaciones y rebajar esa barrera de
entrada, sin sacrificar el rendimiento, se planteó implementar
progresivamente en Julia distintas funcionalidades de HOFEM. Entre las
características del lenguaje que lo hacen adecuado para este propósito
\cite{bezanson2017julia} cabe destacar:

\begin{itemize}
	\item La inferencia de tipos, que permite declararlos o no sin sacrificar 
	en rendimiento, ya que el compilador es capaz de hacerlo del tipo con el que se llama a la función.
	\item El \emph{multiple dispatch}, que selecciona la implementación de una
	función en función de los tipos de todos sus argumentos, de modo que una
	misma rutina sirve tanto para reales como complejos
	\item la utilización de caracteres Unicode que permite mejorar la legibilidad del código como se ve en ~\ref{cod:jacobiano}.
	\item No hace falta vectorizar para aumentar el rendimiento.
	\item Los tipos definidos por el usuario tienen una eficiencia parecida a los del propio lenguaje.
\end{itemize}
\begin{lstlisting}[language=Julia, caption={Construcción de la matriz jacobiana del mapeo isoparamétrico.}, label=cod:jacobiano]
	function compute_jacobian(nodes, pt)
	ξ, η, ζ = pt
	dNdξ = [-(1-η)*(1-ζ), (1-η)*(1-ζ), η*(1-ζ), -η*(1-ζ),
	-(1-η)*ζ,     (1-η)*ζ,     η*ζ,     -η*ζ]
	...
\end{lstlisting}

\section{El problema electromagnético: Maxwell en régimen armónico}\label{sec:maxwell}
% QUÉ ESCRIBIR: de las ecuaciones de Maxwell a la ecuación de doble rotacional
% para el campo eléctrico en una cavidad sin fuentes \cite{jin2015, pozar2012}.
\begin{equation}\label{eq:curlcurl}
	\nabla \times \left( \frac{1}{\mu_r} \nabla \times \mathbf{E} \right)
	- k_0^2\, \varepsilon_r\, \mathbf{E} = \mathbf{0}
	\qquad \text{en } \Omega,
\end{equation}
% con la condición de conductor eléctrico perfecto (PEC) en el contorno:
\begin{equation}\label{eq:pec}
	\hat{\mathbf{n}} \times \mathbf{E} = \mathbf{0}
	\qquad \text{en } \partial\Omega .
\end{equation}
% Explica que k_0 es el número de onda en vacío y que resolver la cavidad
% equivale a un problema de autovalores: los k_0 admisibles son las
% frecuencias de resonancia.

\section{El método de los elementos finitos en electromagnetismo}\label{sec:mef}
% QUÉ ESCRIBIR: formulación débil (Galerkin), discretización, referencias
% generales \cite{jin2015, salazarpalma1998, monk2003}.
\begin{equation}\label{eq:debil}
	\int_{\Omega} \frac{1}{\mu_r} (\nabla \times \mathbf{W})\cdot(\nabla \times \mathbf{E})
	\, d\Omega
	- k_0^2 \int_{\Omega} \varepsilon_r\, \mathbf{W}\cdot\mathbf{E} \, d\Omega = 0 .
\end{equation}

\subsection{Grados de libertad: arista, cara y volumen}\label{sec:dofs}
% QUÉ ESCRIBIR: los 54 funcionales (24 de arista, 24 de cara, 6 de volumen)
% y qué garantiza cada bloque. Los pesos concretos q están en la Tabla I del
% artículo; reprodúcela o remite a ella.
\begin{align}
	g_i^{\text{arista}}(\mathbf{u}) &= \int_{e} (\mathbf{u}\cdot\hat{\mathbf{t}})\, q \; dl, \label{eq:dof-arista}\\
	g_i^{\text{cara}}(\mathbf{u})   &= \int_{f} (\hat{\mathbf{n}}\times\mathbf{u})\cdot \mathbf{q} \; ds, \label{eq:dof-cara}\\
	g_i^{\text{volumen}}(\mathbf{u})&= \int_{H} \mathbf{u}\cdot \mathbf{q} \; dV. \label{eq:dof-volumen}
\end{align}
% Idea clave a explicar: los DOF de arista son los que se COMPARTEN entre
% elementos vecinos y por eso imponen la continuidad tangencial.

\subsection{Generación sistemática de coeficientes: $\mathbf{GW}=\mathbf{I}$}\label{sec:coeficientes}

	\mathbf{G}\,\mathbf{W} = \mathbf{I}_{54}, \qquad
	\mathbf{N}_j = \sum_{k=1}^{54} W_{kj}\, \mathbf{u}_k .
\end{equation}
